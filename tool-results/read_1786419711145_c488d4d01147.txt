    98→  const [result, setResult] = useState<{
    99→    success: boolean;
   100→    imported: number;
   101→    created: number;
   102→    updated: number;

   104→  } | null>(null);
   105→  const [dragOver, setDragOver] = useState(false);
   106→  const [uploadedFileName, setUploadedFileName] = useState<string>("");
   107→
   108→  // OPD selection

   127→

   143→
   144→  const downloadTemplate = async (format: 'xlsx' | 'csv') => {
   145→    try {
   146→      const res = await fetch(`/api/admin/import/template?jenis=${jenis}&format=${format}`, { credentials: 'include' });
   147→      if (!res.ok) throw new Error("Gagal mengunduh template");

   162→      toast({
   163→        title: "Error",

   165→        variant: "destructive",

   168→  };
   169→
   170→  /**
   171→   * Split a CSV line respecting quoted fields.
   172→   * Supports both comma and semicolon as delimiter (auto-detect).
   173→   * Quoted fields (e.g., "31.500,5") are preserved as single values.
   174→   */
   175→  const splitCSVLine = (line: string, delimiter: string): string[] => {
   176→    const result: string[] = [];
   177→    let current = '';
   178→    let inQuotes = false;
   179→
   180→    for (let i = 0; i < line.length; i++) {
   181→      const ch = line[i];
   182→      if (ch === '"') {
   183→        if (inQuotes && i + 1 < line.length && line[i + 1] === '"') {
   184→          // Escaped quote
   185→          current += '"';
   186→          i++;
   187→        } else {
   188→          inQuotes = !inQuotes;
   189→        }
   190→      } else if (ch === delimiter && !inQuotes) {
   191→        result.push(current.trim());
   192→        current = '';
   193→      } else {
   194→        current += ch;
   195→      }
   196→    }
   197→    result.push(current.trim());
   198→    return result;
   199→  };
   200→
   201→  const parseCSV = (text: string): { rows: ImportRow[]; errors: string[] } => {
   202→    const lines = text
   203→      .split("\n")
   204→      .map((l) => l.trim())
   205→      .filter((l) => l && !l.startsWith("#"));
   206→
   207→    if (lines.length === 0) {
   208→      return { rows: [], errors: ["File kosong"] };
   209→    }
   210→
   211→    // Auto-detect delimiter: if first data line has semicolons, use semicolon
   212→    // This is common in Indonesian/European CSV where comma is the decimal separator
   213→    const firstDataLine = lines.find(l => !l.startsWith('#')) || '';
   214→    const delimiter = firstDataLine.includes(';') ? ';' : ',';
   215→
   216→    // Skip header line
   217→    const dataLines = lines.slice(1);
   218→    const rows: ImportRow[] = [];
   219→    const errors: string[] = [];
   220→
   221→    dataLines.forEach((line, idx) => {
   222→      const cols = splitCSVLine(line, delimiter);
   223→      if (cols.length < 5) {
   224→        errors.push(`Baris ${idx + 2}: Kolom tidak lengkap (butuh 5 kolom)`);
   225→        return;
   226→      }
   227→
   228→      const [kodeAkun, namaAkun, kategori, anggaranStr, realisasiStr] = cols;
   229→
   230→      if (!kodeAkun) {
   231→        errors.push(`Baris ${idx + 2}: Kode Akun kosong`);
   232→        return;
   233→      }
   234→      if (!namaAkun) {
   235→        errors.push(`Baris ${idx + 2}: Nama Akun kosong`);
   236→        return;
   237→      }
   238→      if (!kategori) {
   239→        errors.push(`Baris ${idx + 2}: Kategori kosong`);
   240→        return;
   241→      }
   242→
   243→      const anggaran = parseNumber(anggaranStr);
   244→      const realisasi = parseNumber(realisasiStr);
   245→
   246→      if (isNaN(anggaran) || anggaran < 0) {
   247→        errors.push(`Baris ${idx + 2}: Anggaran tidak valid ("${anggaranStr}")`);
   248→        return;
   249→      }
   250→      if (isNaN(realisasi) || realisasi < 0) {
   251→        errors.push(`Baris ${idx + 2}: Realisasi tidak valid ("${realisasiStr}")`);
   252→        return;
   253→      }
   254→
   255→      rows.push({ kodeAkun, namaAkun, kategori, anggaran, realisasi });
   256→    });
   257→
   258→    return { rows, errors };
   259→  };
   260→
   261→  const parseXLSX = (data: ArrayBuffer): { rows: ImportRow[]; errors: string[] } => {
   262→    try {
   263→      const workbook = XLSX.read(data, { type: 'array' });
   264→
   265→      // Try to find the "Template Data" sheet first, otherwise use first sheet
   266→      let sheetName = workbook.SheetNames.find(name =>
   267→        name.toLowerCase().includes('template') || name.toLowerCase().includes('data')
   268→      );
   269→      if (!sheetName) {
   270→        sheetName = workbook.SheetNames[0];
   271→      }
   272→
   273→      const ws = workbook.Sheets[sheetName];
   274→      if (!ws) {
   275→        return { rows: [], errors: ["Sheet tidak ditemukan dalam file"] };
   276→      }
   277→
   278→      // Convert sheet to array of arrays
   279→      const sheetData: unknown[][] = XLSX.utils.sheet_to_json(ws, { header: 1, defval: '' });
   280→
   281→      if (sheetData.length === 0) {
   282→        return { rows: [], errors: ["File kosong"] };
   283→      }
   284→
   285→      // Skip header row (first row)
   286→      const dataRows = sheetData.slice(1);
   287→      const rows: ImportRow[] = [];
   288→      const errors: string[] = [];
   289→
   290→      dataRows.forEach((row, idx) => {
   291→        // Skip empty rows (but keep rows where anggaran/realisasi is 0)
   292→        if (!row || row.length === 0 || row.every(cell => cell == null || cell === '')) {
   293→          return;
   294→        }
   295→
   296→        const kodeAkun = String(row[0] ?? '').trim();
   297→        const namaAkun = String(row[1] ?? '').trim();
   298→        const kategori = String(row[2] ?? '').trim();
   299→        // Use ?? instead of || to preserve 0 values (0 || '' evaluates to '')
   300→        const anggaranStr = String(row[3] ?? '').trim();
   301→        const realisasiStr = String(row[4] ?? '').trim();
   302→
   303→        // Skip rows with no kode akun (likely empty template rows)
   304→        if (!kodeAkun) return;
   305→
   306→        if (!namaAkun) {
   307→          errors.push(`Baris ${idx + 2}: Nama Akun kosong`);
   308→          return;
   309→        }
   310→        if (!kategori) {
   311→          errors.push(`Baris ${idx + 2}: Kategori kosong`);
   312→          return;
   313→        }
   314→
   315→        const anggaran = parseNumber(anggaranStr);
   316→        const realisasi = parseNumber(realisasiStr);
   317→
   318→        if (isNaN(anggaran) || anggaran < 0) {
   319→          errors.push(`Baris ${idx + 2}: Anggaran tidak valid ("${anggaranStr}")`);
   320→          return;
   321→        }
   322→        if (isNaN(realisasi) || realisasi < 0) {
   323→          errors.push(`Baris ${idx + 2}: Realisasi tidak valid ("${realisasiStr}")`);
   324→          return;
   325→        }
   326→
   327→        rows.push({ kodeAkun, namaAkun, kategori, anggaran, realisasi });
   328→      });
   329→
   330→      return { rows, errors };
   331→    } catch (err) {
   332→      return { rows: [], errors: [`Gagal membaca file XLSX: ${err instanceof Error ? err.message : 'Unknown error'}`] };
   333→    }
   334→  };
   335→
   336→  const parseNumber = (str: string): number => {
   337→    if (!str || !str.trim()) return NaN;
   338→    // Remove "Rp" / "Rp." prefix and whitespace
   339→    let cleaned = str.replace(/Rp\.?\s*/gi, '').trim();
   340→    // Remove dots as thousand separators (Indonesian format: 31.500.000.000)
   341→    cleaned = cleaned.replace(/\./g, '');
   342→    // Replace comma with dot as decimal separator (Indonesian format: 31500,5 → 31500.5)
   343→    cleaned = cleaned.replace(/,/g, '.');
   344→    const num = parseFloat(cleaned);
   345→    return num;
   346→  };
   347→
   348→  const handleFile = (file: File) => {
   349→    const isXLSX = file.name.endsWith('.xlsx') || file.name.endsWith('.xls');
   350→    const isCSV = file.name.endsWith('.csv') || file.name.endsWith('.txt');
   351→
   352→    if (!isXLSX && !isCSV) {
   353→      toast({
   354→        title: "Format Salah",
   355→        description: "Hanya file XLSX (Excel) dan CSV yang didukung",
   356→        variant: "destructive",
   357→      });
   358→      return;
   359→    }
   360→
   361→    setUploadedFileName(file.name);
   362→
   363→    if (isXLSX) {
   364→      const reader = new FileReader();
   365→      reader.onload = (e) => {
   366→        const data = e.target?.result as ArrayBuffer;
   367→        const { rows, errors } = parseXLSX(data);
   368→        setParseErrors(errors);
   369→        setParsedRows(rows);
   370→
   371→        if (rows.length > 0) {
   372→          setStep("preview");
   373→        } else if (errors.length > 0) {
   374→          toast({
   375→            title: "Gagal Membaca File",
   376→            description: "Tidak ada data valid yang dapat diimpor",
   377→            variant: "destructive",
   378→          });
   379→        } else {
   380→          toast({
   381→            title: "File Kosong",
   382→            description: "Tidak ada data ditemukan dalam file Excel",
   383→            variant: "destructive",
   384→          });
   385→        }
   386→      };
   387→      reader.readAsArrayBuffer(file);
   388→    } else {
   389→      // CSV
   390→      const reader = new FileReader();
   391→      reader.onload = (e) => {
   392→        const text = e.target?.result as string;
   393→        const { rows, errors } = parseCSV(text);
   394→        setParseErrors(errors);
   395→        setParsedRows(rows);
   396→
   397→        if (rows.length > 0) {
   398→          setStep("preview");
   399→        } else if (errors.length > 0) {
   400→          toast({
   401→            title: "Gagal Membaca File",
   402→            description: "Tidak ada data valid yang dapat diimpor",
   403→            variant: "destructive",
   404→          });
   405→        }
   406→      };
   407→      reader.readAsText(file);
   408→    }
   409→  };
   410→
   411→  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   412→    const file = e.target.files?.[0];
   413→    if (file) handleFile(file);
   414→    // Reset the input so the same file can be re-selected
   415→    e.target.value = "";
   416→  };
   417→
   418→  const handleDrop = (e: React.DragEvent) => {
   419→    e.preventDefault();
   420→    setDragOver(false);
   421→    const file = e.dataTransfer.files?.[0];
   422→    if (file) handleFile(file);
   423→  };
   424→
   425→  const handleDragOver = (e: React.DragEvent) => {
   426→    e.preventDefault();
   427→    setDragOver(true);
   428→  };
   429→
   430→  const handleDragLeave = () => {
   431→    setDragOver(false);
   432→  };
   433→
   434→  const removeRow = (index: number) => {
   435→    setParsedRows((prev) => prev.filter((_, i) => i !== index));
   436→  };
   437→
   438→  const handleImport = async () => {
   439→    if (!tahunAnggaranId || parsedRows.length === 0) return;
   440→
   441→    setImporting(true);
   442→    try {
   443→      const res = await fetch("/api/admin/import", {
   444→        method: "POST",
   445→        headers: { "Content-Type": "application/json" },
   446→        credentials: "include",
   447→        body: JSON.stringify({
   448→          jenis,
   449→          tahunAnggaranId,
   450→          rows: parsedRows,
   451→          mode,
   452→          opdId: selectedOpdId && selectedOpdId !== "__none__" ? selectedOpdId : null,
   453→        }),
   454→      });
   455→
   456→      const json = await res.json();
   457→
   458→      if (!res.ok) {
   459→        if (json.validationErrors) {
   460→          const errMsg = json.validationErrors
   461→            .slice(0, 5)
   462→            .map((v: { row: number; error: string }) => `Baris ${v.row}: ${v.error}`)
   463→            .join("\n");
   464→          throw new Error(`Validasi gagal:\n${errMsg}`);
   465→        }
   466→        throw new Error(json.error || "Gagal mengimpor data");
   467→      }
   468→
   469→      setResult({
   470→        success: true,
   471→        imported: json.imported,
   472→        created: json.created || 0,
   473→        updated: json.updated || 0,
   474→        message: json.message,
   475→      });
   476→      setStep("result");
   477→      onSuccess();
   478→    } catch (err) {
   479→      toast({
   480→        title: "Gagal Import",
   481→        description: err instanceof Error ? err.message : "Terjadi kesalahan",
   482→        variant: "destructive",
   483→      });
   484→      setResult({
   485→        success: false,
   486→        imported: 0,
   487→        created: 0,
   488→        updated: 0,
   489→        message: err instanceof Error ? err.message : "Terjadi kesalahan",
   490→      });
   491→      setStep("result");
   492→    } finally {
   493→      setImporting(false);
   494→    }
   495→  };
   496→
   497→  const formatRupiah = (value: number) => {
   498→    if (value >= 1_000_000_000_000) return `${(value / 1_000_000_000_000).toFixed(1)} T`;
   499→    if (value >= 1_000_000_000) return `${(value / 1_000_000_000).toFixed(1)} M`;
   500→    if (value >= 1_000_000) return `${(value / 1_000_000).toFixed(1)} Jt`;
   501→    return value.toLocaleString("id-ID");
   502→  };
   503→
   504→  return (
   505→    <Dialog open={open} onOpenChange={handleClose}>
   506→      <DialogContent className="sm:max-w-2xl max-h-[85vh]">
   507→        <DialogHeader>
   508→          <DialogTitle className="flex items-center gap-2">
   509→            <div className={`w-2.5 h-2.5 rounded-full ${jenisColors[jenis]}`} />
   510→            Import Data {jenisLabels[jenis]}
   511→          </DialogTitle>
   512→          <DialogDescription>
   513→            Impor data {jenisLabels[jenis]} dari file Excel (.xlsx) atau CSV
   514→          </DialogDescription>
   515→        </DialogHeader>
   516→
   517→        <AnimatePresence mode="wait">
   518→          {/* ====== STEP 1: UPLOAD ====== */}
   519→          {step === "upload" && (
   520→            <motion.div
   521→              key="upload"
   522→              initial={{ opacity: 0, x: -20 }}
   523→              animate={{ opacity: 1, x: 0 }}
   524→              exit={{ opacity: 0, x: 20 }}
   525→              className="space-y-4"
   526→            >
   527→              {/* OPD Selection for Admin/Superadmin */}
   528→              {isAdmin && (
   529→                <div className="space-y-2">
   530→                  <Label className="flex items-center gap-2 text-sm font-medium">
   531→                    <Building2 className="w-4 h-4" />
   532→                    OPD Tujuan Import
   533→                  </Label>
   534→                  <div className="flex items-center gap-2 mb-1">
   535→                    <Badge variant="outline" className="text-xs">
   536→                      <ShieldCheck className="w-3 h-3 mr-1" />
   537→                      {user?.role === "superadmin" ? "Super Admin" : "Admin"} — dapat memilih OPD
   538→                    </Badge>
   539→                  </div>
   540→                  {opdLoading ? (
   541→                    <div className="flex items-center gap-2 text-sm text-muted-foreground">
   542→                      <Loader2 className="w-4 h-4 animate-spin" />
   543→                      Memuat daftar OPD...
   544→                    </div>
   545→                  ) : (
   546→                    <Select value={selectedOpdId} onValueChange={setSelectedOpdId}>
   547→                      <SelectTrigger>
   548→                        <SelectValue placeholder="-- Tanpa OPD (Data Global) --" />
   549→                      </SelectTrigger>
   550→                      <SelectContent>
   551→                        <SelectItem value="__none__">-- Tanpa OPD (Data Global) --</SelectItem>
   552→                        {opdList.map(opd => (
   553→                          <SelectItem key={opd.id} value={opd.id}>
   554→                            {opd.kodeOpd} — {opd.namaOpd}
   555→                          </SelectItem>
   556→                        ))}
   557→                      </SelectContent>
   558→                    </Select>
   559→                  )}
   560→                </div>
   561→              )}
   562→
   563→              {/* OPD User Info */}
   564→              {isOpdUser && (
   565→                <div className="p-3 rounded-lg border border-amber-200 bg-amber-50 dark:bg-amber-950/20 dark:border-amber-800">
   566→                  <div className="flex items-center gap-2 text-sm font-medium text-amber-700 dark:text-amber-400">
   567→                    <ShieldX className="w-4 h-4" />
   568→                    OPD User — Data hanya dapat diimpor ke OPD Anda sendiri
   569→                  </div>
   570→                </div>
   571→              )}
   572→
   573→              {/* Download template - Professional dual format card */}
   574→              <div className="rounded-xl border bg-gradient-to-br from-muted/30 to-muted/10 overflow-hidden">
   575→                <div className={`px-4 py-2.5 bg-gradient-to-r ${jenisGradients[jenis]}`}>
