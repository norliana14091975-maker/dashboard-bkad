   182→   182→  /**
   183→   183→   * Split a CSV line respecting quoted fields.
   184→   184→   * Supports both comma and semicolon as delimiter (auto-detect).

   187→   187→  const splitCSVLine = (line: string, delimiter: string): string[] => {
   188→   188→    const result: string[] = [];
   189→   189→    let current = '';
   190→   190→    let inQuotes = false;
   191→   191→
   192→   192→    for (let i = 0; i < line.length; i++) {
   193→   193→      const ch = line[i];
   194→   194→      if (ch === '"') {
   195→   195→        if (inQuotes && i + 1 < line.length && line[i + 1] === '"') {
   196→   196→          // Escaped quote
   197→   197→          current += '"';
   198→   198→          i++;
   199→   199→        } else {
   200→   200→          inQuotes = !inQuotes;
   201→   201→        }
   202→   202→      } else if (ch === delimiter && !inQuotes) {
   203→   203→        result.push(current.trim());
   204→   204→        current = '';
   205→   205→      } else {
   206→   206→        current += ch;
   207→   207→      }
   208→   208→    }
   209→   209→    result.push(current.trim());
   210→   210→    return result;
   211→   211→  };
   212→   212→
   213→   213→  const parseCSV = (text: string): { rows: ImportRow[]; errors: string[] } => {
   214→   214→    const lines = text
   215→   215→      .split("\n")
   216→   216→      .map((l) => l.trim())
   217→   217→      .filter((l) => l && !l.startsWith("#"));
   218→   218→
   219→   219→    if (lines.length === 0) {
   220→   220→      return { rows: [], errors: ["File kosong"] };
   221→   221→    }
   222→   222→
   223→   223→    // Auto-detect delimiter: if first data line has semicolons, use semicolon
   224→   224→    // This is common in Indonesian/European CSV where comma is the decimal separator
   225→   225→    const firstDataLine = lines.find(l => !l.startsWith('#')) || '';
   226→   226→    const delimiter = firstDataLine.includes(';') ? ';' : ',';
   227→   227→
   228→   228→    // Skip header line
   229→   229→    const dataLines = lines.slice(1);
   230→   230→    const rows: ImportRow[] = [];
   231→   231→    const errors: string[] = [];
   232→   232→
   233→   233→    dataLines.forEach((line, idx) => {
   234→   234→      const cols = splitCSVLine(line, delimiter);
   235→   235→      if (cols.length < 5) {
   236→   236→        errors.push(`Baris ${idx + 2}: Kolom tidak lengkap (butuh 5 kolom)`);
   237→   237→        return;
   238→   238→      }
   239→   239→
   240→   240→      const [kodeAkun, namaAkun, kategori, anggaranStr, realisasiStr] = cols;
   241→   241→
   242→   242→      if (!kodeAkun) {
   243→   243→        errors.push(`Baris ${idx + 2}: Kode Akun kosong`);
   244→   244→        return;
   245→   245→      }
   246→   246→      if (!namaAkun) {
   247→   247→        errors.push(`Baris ${idx + 2}: Nama Akun kosong`);
   248→   248→        return;
   249→   249→      }
   250→   250→      if (!kategori) {
   251→   251→        errors.push(`Baris ${idx + 2}: Kategori kosong`);
   252→   252→        return;
   253→   253→      }
   254→   254→
   255→   255→      const anggaran = parseNumber(anggaranStr);
   256→   256→      const realisasi = parseNumber(realisasiStr);
   257→   257→
   258→   258→      if (isNaN(anggaran) || anggaran < 0) {
   259→   259→        errors.push(`Baris ${idx + 2}: Anggaran tidak valid ("${anggaranStr}")`);
   260→   260→        return;
   261→   261→      }
   262→   262→      if (isNaN(realisasi) || realisasi < 0) {
   263→   263→        errors.push(`Baris ${idx + 2}: Realisasi tidak valid ("${realisasiStr}")`);
   264→   264→        return;
   265→   265→      }
   266→   266→
   267→   267→      rows.push({ kodeAkun, namaAkun, kategori, anggaran, realisasi });
   268→   268→    });
   269→   269→
   270→   270→    return { rows, errors };
   271→   271→  };

   273→   273→  const parseXLSX = (data: ArrayBuffer): { rows: ImportRow[]; errors: string[] } => {
   274→   274→    try {
   275→   275→      const workbook = XLSX.read(data, { type: 'array' });
   276→   276→
   277→   277→      // Try to find the "Template Data" sheet first, otherwise use first sheet
   278→   278→      let sheetName = workbook.SheetNames.find(name =>
   279→   279→        name.toLowerCase().includes('template') || name.toLowerCase().includes('data')
   280→   280→      );
   281→   281→      if (!sheetName) {
   282→   282→        sheetName = workbook.SheetNames[0];
   283→   283→      }
   284→   284→
   285→   285→      const ws = workbook.Sheets[sheetName];
   286→   286→      if (!ws) {
   287→   287→        return { rows: [], errors: ["Sheet tidak ditemukan dalam file"] };
   288→   288→      }
   289→   289→
   290→   290→      // Convert sheet to array of arrays
   291→   291→      const sheetData: unknown[][] = XLSX.utils.sheet_to_json(ws, { header: 1, defval: '' });
   292→   292→
   293→   293→      if (sheetData.length === 0) {
   294→   294→        return { rows: [], errors: ["File kosong"] };
   295→   295→      }
   296→   296→
   297→   297→      // Skip header row (first row)
   298→   298→      const dataRows = sheetData.slice(1);
   299→   299→      const rows: ImportRow[] = [];
   300→   300→      const errors: string[] = [];
   301→   301→
   302→   302→      dataRows.forEach((row, idx) => {
   303→   303→        // Skip empty rows (but keep rows where anggaran/realisasi is 0)
   304→   304→        if (!row || row.length === 0 || row.every(cell => cell == null || cell === '')) {
   305→   305→          return;
   306→   306→        }
   307→   307→
   308→   308→        const kodeAkun = String(row[0] ?? '').trim();
   309→   309→        const namaAkun = String(row[1] ?? '').trim();
   310→   310→        const kategori = String(row[2] ?? '').trim();
   311→   311→        // Use ?? instead of || to preserve 0 values (0 || '' evaluates to '')
   312→   312→        const anggaranStr = String(row[3] ?? '').trim();
   313→   313→        const realisasiStr = String(row[4] ?? '').trim();
   314→   314→
   315→   315→        // Skip rows with no kode akun (likely empty template rows)
   316→   316→        if (!kodeAkun) return;
   317→   317→
   318→   318→        if (!namaAkun) {
   319→   319→          errors.push(`Baris ${idx + 2}: Nama Akun kosong`);
   320→   320→          return;
   321→   321→        }
   322→   322→        if (!kategori) {
   323→   323→          errors.push(`Baris ${idx + 2}: Kategori kosong`);
   324→   324→          return;
   325→   325→        }
   326→   326→
   327→   327→        const anggaran = parseNumber(anggaranStr);
   328→   328→        const realisasi = parseNumber(realisasiStr);
   329→   329→
   330→   330→        if (isNaN(anggaran) || anggaran < 0) {
   331→   331→          errors.push(`Baris ${idx + 2}: Anggaran tidak valid ("${anggaranStr}")`);
   332→   332→          return;
   333→   333→        }
   334→   334→        if (isNaN(realisasi) || realisasi < 0) {
   335→   335→          errors.push(`Baris ${idx + 2}: Realisasi tidak valid ("${realisasiStr}")`);
   336→   336→          return;
   337→   337→        }
   338→   338→
   339→   339→        rows.push({ kodeAkun, namaAkun, kategori, anggaran, realisasi });
   340→   340→      });
   341→   341→
   342→   342→      return { rows, errors };
   343→   343→    } catch (err) {
   344→   344→      return { rows: [], errors: [`Gagal membaca file XLSX: ${err instanceof Error ? err.message : 'Unknown error'}`] };
   345→   345→    }
   346→   346→  };
   347→   347→
   348→   348→  const parseNumber = (str: string): number => {
   349→   349→    if (!str || !str.trim()) return NaN;
   350→   350→    // Remove "Rp" / "Rp." prefix and whitespace
   351→   351→    let cleaned = str.replace(/Rp\.?\s*/gi, '').trim();
   352→   352→    // Remove dots as thousand separators (Indonesian format: 31.500.000.000)
   353→   353→    cleaned = cleaned.replace(/\./g, '');
   354→   354→    // Replace comma with dot as decimal separator (Indonesian format: 31500,5 → 31500.5)
   355→   355→    cleaned = cleaned.replace(/,/g, '.');
   356→   356→    const num = parseFloat(cleaned);
   357→   357→    return num;
   358→   358→  };
   359→   359→
   360→   360→  // Auto-detect jenis and kategori from kode akun prefix
   361→   361→  const enrichRowsWithAutoDetect = (rows: ImportRow[]): ImportRow[] => {
   362→   362→    if (!isAutoDetect) return rows;
   363→   363→    return rows.map(row => {
   364→   364→      const firstDigit = row.kodeAkun.trim().charAt(0);
   365→   365→      let detectedJenis: JenisData | null = null;
   366→   366→      let detectedKategori = row.kategori; // keep existing kategori from file
   367→   367→
   368→   368→      switch (firstDigit) {
   369→   369→        case '4':
   370→   370→          detectedJenis = 'pendapatan';
   371→   371→          if (!row.kategori) {
   372→   372→            const seg2 = row.kodeAkun.split('.')[1];
   373→   373→            detectedKategori = seg2 === '1' ? 'PAD' : seg2 === '2' ? 'Transfer' : seg2 === '3' ? 'Lain-Lain' : 'PAD';
   374→   374→          }
   375→   375→          break;
   376→   376→        case '5':
   377→   377→          detectedJenis = 'belanja';
   378→   378→          if (!row.kategori) {
   379→   379→            const seg2 = row.kodeAkun.split('.')[1];
   380→   380→            detectedKategori = seg2 === '1' ? 'Operasi' : seg2 === '2' ? 'Modal' : seg2 === '3' ? 'Tidak Terduga' : seg2 === '4' ? 'Transfer' : 'Operasi';
   381→   381→          }
   382→   382→          break;
   383→   383→        case '6':
   384→   384→          detectedJenis = 'pembiayaan';
   385→   385→          if (!row.kategori) {
   386→   386→            const seg2 = row.kodeAkun.split('.')[1];
   387→   387→            detectedKategori = seg2 === '1' ? 'Penerimaan' : seg2 === '2' ? 'Pengeluaran' : 'Penerimaan';
   388→   388→          }
   389→   389→          break;
   390→   390→      }
   391→   391→
   392→   392→      return {
   393→   393→        ...row,
   394→   394→        kategori: detectedKategori,
   395→   395→        jenis: detectedJenis || undefined,
   396→   396→        detectedJenis,
   397→   397→        detectedKategori,
   398→   398→      };
   399→   399→    });
   400→   400→  };

   402→   402→  const handleFile = (file: File) => {
   403→   403→    const isXLSX = file.name.endsWith('.xlsx') || file.name.endsWith('.xls');

   428→   428→          toast({
   429→   429→            title: "Gagal Membaca File",

   431→   431→            variant: "destructive",
   432→   432→          });
   433→   433→        } else {
   434→   434→          toast({
   435→   435→            title: "File Kosong",

   437→   437→            variant: "destructive",
   438→   438→          });
   439→   439→        }
   440→   440→      };
   441→   441→      reader.readAsArrayBuffer(file);
   442→   442→    } else {
   443→   443→      // CSV
   444→   444→      const reader = new FileReader();
   445→   445→      reader.onload = (e) => {
   446→   446→        const text = e.target?.result as string;
   447→   447→        const { rows, errors } = parseCSV(text);
   448→   448→        setParseErrors(errors);
   449→   449→        setParsedRows(enrichRowsWithAutoDetect(rows));
   450→   450→
   451→   451→        if (rows.length > 0) {
   452→   452→          setStep("preview");
   453→   453→        } else if (errors.length > 0) {
   454→   454→          toast({
   455→   455→            title: "Gagal Membaca File",
   456→   456→            description: "Tidak ada data valid yang dapat diimpor",
   457→   457→            variant: "destructive",
   458→   458→          });
   459→   459→        }
   460→   460→      };
   461→   461→      reader.readAsText(file);
   462→   462→    }
   463→   463→  };
