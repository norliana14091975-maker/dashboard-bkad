   136→   136→  /**
   137→   137→   * Split a CSV line respecting quoted fields.
   138→   138→   * Supports both comma and semicolon as delimiter (auto-detect).
   139→   139→   * Quoted fields (e.g., "31.500,5") are preserved as single values.
   140→   140→   */
   141→   141→  const splitCSVLine = (line: string, delimiter: string): string[] => {
   142→   142→    const result: string[] = [];
   143→   143→    let current = '';
   144→   144→    let inQuotes = false;
   145→   145→
   146→   146→    for (let i = 0; i < line.length; i++) {
   147→   147→      const ch = line[i];
   148→   148→      if (ch === '"') {
   149→   149→        if (inQuotes && i + 1 < line.length && line[i + 1] === '"') {
   150→   150→          // Escaped quote
   151→   151→          current += '"';
   152→   152→          i++;
   153→   153→        } else {
   154→   154→          inQuotes = !inQuotes;
   155→   155→        }
   156→   156→      } else if (ch === delimiter && !inQuotes) {
   157→   157→        result.push(current.trim());
   158→   158→        current = '';
   159→   159→      } else {
   160→   160→        current += ch;
   161→   161→      }
   162→   162→    }
   163→   163→    result.push(current.trim());
   164→   164→    return result;
   165→   165→  };
   166→   166→
   167→   167→  const parseCSV = (text: string): { rows: ImportRow[]; errors: string[] } => {
   168→   168→    const lines = text
   169→   169→      .split("\n")
   170→   170→      .map((l) => l.trim())
   171→   171→      .filter((l) => l && !l.startsWith("#"));
   172→   172→
   173→   173→    if (lines.length === 0) {
   174→   174→      return { rows: [], errors: ["File kosong"] };
   175→   175→    }
   176→   176→
   177→   177→    // Auto-detect delimiter: if first data line has semicolons, use semicolon
   178→   178→    // This is common in Indonesian/European CSV where comma is the decimal separator
   179→   179→    const firstDataLine = lines.find(l => !l.startsWith('#')) || '';
   180→   180→    const delimiter = firstDataLine.includes(';') ? ';' : ',';
   181→   181→
   182→   182→    // Skip header line
   183→   183→    const dataLines = lines.slice(1);
   184→   184→    const rows: ImportRow[] = [];
   185→   185→    const errors: string[] = [];
   186→   186→
   187→   187→    dataLines.forEach((line, idx) => {
   188→   188→      const cols = splitCSVLine(line, delimiter);
   189→   189→      if (cols.length < 5) {
   190→   190→        errors.push(`Baris ${idx + 2}: Kolom tidak lengkap (butuh 5 kolom)`);
   191→   191→        return;
   192→   192→      }
   193→   193→
   194→   194→      const [kodeAkun, namaAkun, kategori, anggaranStr, realisasiStr] = cols;
   195→   195→
   196→   196→      if (!kodeAkun) {
   197→   197→        errors.push(`Baris ${idx + 2}: Kode Akun kosong`);
   198→   198→        return;
   199→   199→      }
   200→   200→      if (!namaAkun) {
   201→   201→        errors.push(`Baris ${idx + 2}: Nama Akun kosong`);
   202→   202→        return;
   203→   203→      }
   204→   204→      if (!kategori) {
   205→   205→        errors.push(`Baris ${idx + 2}: Kategori kosong`);
   206→   206→        return;
   207→   207→      }
   208→   208→
   209→   209→      const anggaran = parseNumber(anggaranStr);
   210→   210→      const realisasi = parseNumber(realisasiStr);
   211→   211→
   212→   212→      if (isNaN(anggaran) || anggaran < 0) {
   213→   213→        errors.push(`Baris ${idx + 2}: Anggaran tidak valid ("${anggaranStr}")`);
   214→   214→        return;
   215→   215→      }
   216→   216→      if (isNaN(realisasi) || realisasi < 0) {
   217→   217→        errors.push(`Baris ${idx + 2}: Realisasi tidak valid ("${realisasiStr}")`);
   218→   218→        return;
   219→   219→      }
   220→   220→
   221→   221→      rows.push({ kodeAkun, namaAkun, kategori, anggaran, realisasi });
   222→   222→    });
   223→   223→
   224→   224→    return { rows, errors };
   225→   225→  };
   226→   226→
   227→   227→  const parseXLSX = (data: ArrayBuffer): { rows: ImportRow[]; errors: string[] } => {
   228→   228→    try {
   229→   229→      const workbook = XLSX.read(data, { type: 'array' });
   230→   230→
   231→   231→      // Try to find the "Template Data" sheet first, otherwise use first sheet
   232→   232→      let sheetName = workbook.SheetNames.find(name =>
   233→   233→        name.toLowerCase().includes('template') || name.toLowerCase().includes('data')
   234→   234→      );
   235→   235→      if (!sheetName) {
   236→   236→        sheetName = workbook.SheetNames[0];
   237→   237→      }
   238→   238→
   239→   239→      const ws = workbook.Sheets[sheetName];
   240→   240→      if (!ws) {
   241→   241→        return { rows: [], errors: ["Sheet tidak ditemukan dalam file"] };
   242→   242→      }
   243→   243→
   244→   244→      // Convert sheet to array of arrays
   245→   245→      const sheetData: unknown[][] = XLSX.utils.sheet_to_json(ws, { header: 1, defval: '' });
   246→   246→
   247→   247→      if (sheetData.length === 0) {
   248→   248→        return { rows: [], errors: ["File kosong"] };
   249→   249→      }
   250→   250→
   251→   251→      // Skip header row (first row)
   252→   252→      const dataRows = sheetData.slice(1);
   253→   253→      const rows: ImportRow[] = [];
   254→   254→      const errors: string[] = [];
   255→   255→
   256→   256→      dataRows.forEach((row, idx) => {
   257→   257→        // Skip empty rows (but keep rows where anggaran/realisasi is 0)
   258→   258→        if (!row || row.length === 0 || row.every(cell => cell == null || cell === '')) {
   259→   259→          return;
   260→   260→        }
   261→   261→
   262→   262→        const kodeAkun = String(row[0] ?? '').trim();
   263→   263→        const namaAkun = String(row[1] ?? '').trim();
   264→   264→        const kategori = String(row[2] ?? '').trim();
   265→   265→        // Use ?? instead of || to preserve 0 values (0 || '' evaluates to '')
   266→   266→        const anggaranStr = String(row[3] ?? '').trim();
   267→   267→        const realisasiStr = String(row[4] ?? '').trim();
   268→   268→
   269→   269→        // Skip rows with no kode akun (likely empty template rows)
   270→   270→        if (!kodeAkun) return;
   271→   271→
   272→   272→        if (!namaAkun) {
   273→   273→          errors.push(`Baris ${idx + 2}: Nama Akun kosong`);
   274→   274→          return;
   275→   275→        }
   276→   276→        if (!kategori) {
   277→   277→          errors.push(`Baris ${idx + 2}: Kategori kosong`);
   278→   278→          return;
   279→   279→        }
   280→   280→
   281→   281→        const anggaran = parseNumber(anggaranStr);
   282→   282→        const realisasi = parseNumber(realisasiStr);
   283→   283→
   284→   284→        if (isNaN(anggaran) || anggaran < 0) {
   285→   285→          errors.push(`Baris ${idx + 2}: Anggaran tidak valid ("${anggaranStr}")`);
   286→   286→          return;
   287→   287→        }
   288→   288→        if (isNaN(realisasi) || realisasi < 0) {
   289→   289→          errors.push(`Baris ${idx + 2}: Realisasi tidak valid ("${realisasiStr}")`);
   290→   290→          return;
   291→   291→        }
   292→   292→
   293→   293→        rows.push({ kodeAkun, namaAkun, kategori, anggaran, realisasi });
   294→   294→      });
   295→   295→
   296→   296→      return { rows, errors };
   297→   297→    } catch (err) {
   298→   298→      return { rows: [], errors: [`Gagal membaca file XLSX: ${err instanceof Error ? err.message : 'Unknown error'}`] };
   299→   299→    }
   300→   300→  };
   301→   301→
   302→   302→  const parseNumber = (str: string): number => {
   303→   303→    if (!str || !str.trim()) return NaN;
   304→   304→    // Remove "Rp" / "Rp." prefix and whitespace
   305→   305→    let cleaned = str.replace(/Rp\.?\s*/gi, '').trim();
   306→   306→    // Remove dots as thousand separators (Indonesian format: 31.500.000.000)
   307→   307→    cleaned = cleaned.replace(/\./g, '');
   308→   308→    // Replace comma with dot as decimal separator (Indonesian format: 31500,5 → 31500.5)
   309→   309→    cleaned = cleaned.replace(/,/g, '.');
   310→   310→    const num = parseFloat(cleaned);
   311→   311→    return num;
   312→   312→  };
   313→   313→
   314→   314→  const handleFile = (file: File) => {
   315→   315→    const isXLSX = file.name.endsWith('.xlsx') || file.name.endsWith('.xls');
   316→   316→    const isCSV = file.name.endsWith('.csv') || file.name.endsWith('.txt');
   317→   317→
   318→   318→    if (!isXLSX && !isCSV) {
   319→   319→      toast({
   320→   320→        title: "Format Salah",
   321→   321→        description: "Hanya file XLSX (Excel) dan CSV yang didukung",
   322→   322→        variant: "destructive",
   323→   323→      });
   324→   324→      return;
   325→   325→    }
   326→   326→
   327→   327→    setUploadedFileName(file.name);
   328→   328→
   329→   329→    if (isXLSX) {
   330→   330→      const reader = new FileReader();
   331→   331→      reader.onload = (e) => {
   332→   332→        const data = e.target?.result as ArrayBuffer;
   333→   333→        const { rows, errors } = parseXLSX(data);
   334→   334→        setParseErrors(errors);
   335→   335→        setParsedRows(rows);
   336→   336→
   337→   337→        if (rows.length > 0) {
   338→   338→          setStep("preview");
   339→   339→        } else if (errors.length > 0) {
   340→   340→          toast({
   341→   341→            title: "Gagal Membaca File",
   342→   342→            description: "Tidak ada data valid yang dapat diimpor",
   343→   343→            variant: "destructive",
   344→   344→          });
   345→   345→        } else {
   346→   346→          toast({
   347→   347→            title: "File Kosong",
   348→   348→            description: "Tidak ada data ditemukan dalam file Excel",
   349→   349→            variant: "destructive",
   350→   350→          });
   351→   351→        }
   352→   352→      };
   353→   353→      reader.readAsArrayBuffer(file);
   354→   354→    } else {
   355→   355→      // CSV
   356→   356→      const reader = new FileReader();
   357→   357→      reader.onload = (e) => {
   358→   358→        const text = e.target?.result as string;
   359→   359→        const { rows, errors } = parseCSV(text);
   360→   360→        setParseErrors(errors);
   361→   361→        setParsedRows(rows);
   362→   362→
   363→   363→        if (rows.length > 0) {
   364→   364→          setStep("preview");
   365→   365→        } else if (errors.length > 0) {
   366→   366→          toast({
   367→   367→            title: "Gagal Membaca File",
   368→   368→            description: "Tidak ada data valid yang dapat diimpor",
   369→   369→            variant: "destructive",
   370→   370→          });
   371→   371→        }
   372→   372→      };
   373→   373→      reader.readAsText(file);
   374→   374→    }
   375→   375→  };
   376→   376→
   377→   377→  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   378→   378→    const file = e.target.files?.[0];
   379→   379→    if (file) handleFile(file);

   382→   382→  };

   384→   384→  const handleDrop = (e: React.DragEvent) => {
   385→   385→    e.preventDefault();
   386→   386→    setDragOver(false);
   387→   387→    const file = e.dataTransfer.files?.[0];
   388→   388→    if (file) handleFile(file);
   389→   389→  };
   390→   390→
   391→   391→  const handleDragOver = (e: React.DragEvent) => {
   392→   392→    e.preventDefault();
   393→   393→    setDragOver(true);
   394→   394→  };
   395→   395→
   396→   396→  const handleDragLeave = () => {
   397→   397→    setDragOver(false);
   398→   398→  };
   399→   399→
   400→   400→  const removeRow = (index: number) => {
   401→   401→    setParsedRows((prev) => prev.filter((_, i) => i !== index));
   402→   402→  };
   403→   403→
   404→   404→  const handleImport = async () => {
   405→   405→    if (!tahunAnggaranId || parsedRows.length === 0) return;
   406→   406→
   407→   407→    setImporting(true);
   408→   408→    try {
   409→   409→      const res = await fetch("/api/admin/import", {
   410→   410→        method: "POST",
   411→   411→        headers: { "Content-Type": "application/json" },
   412→   412→        credentials: "include",
   413→   413→        body: JSON.stringify({
   414→   414→          jenis,
   415→   415→          tahunAnggaranId,
   416→   416→          rows: parsedRows,
   417→   417→          mode,
   418→   418→        }),
   419→   419→      });
   420→   420→
   421→   421→      const json = await res.json();
   422→   422→
   423→   423→      if (!res.ok) {
   424→   424→        if (json.validationErrors) {
   425→   425→          const errMsg = json.validationErrors
   426→   426→            .slice(0, 5)
   427→   427→            .map((v: { row: number; error: string }) => `Baris ${v.row}: ${v.error}`)
   428→   428→            .join("\n");
   429→   429→          throw new Error(`Validasi gagal:\n${errMsg}`);
   430→   430→        }
   431→   431→        throw new Error(json.error || "Gagal mengimpor data");
   432→   432→      }
   433→   433→
   434→   434→      setResult({
   435→   435→        success: true,
   436→   436→        imported: json.imported,
   437→   437→        created: json.created || 0,
   438→   438→        updated: json.updated || 0,
   439→   439→        message: json.message,
   440→   440→      });
   441→   441→      setStep("result");
   442→   442→      onSuccess();
   443→   443→    } catch (err) {
   444→   444→      toast({
   445→   445→        title: "Gagal Import",
   446→   446→        description: err instanceof Error ? err.message : "Terjadi kesalahan",
   447→   447→        variant: "destructive",
   448→   448→      });
   449→   449→      setResult({
   450→   450→        success: false,
   451→   451→        imported: 0,
   452→   452→        created: 0,
   453→   453→        updated: 0,
   454→   454→        message: err instanceof Error ? err.message : "Terjadi kesalahan",
   455→   455→      });
   456→   456→      setStep("result");
   457→   457→    } finally {
   458→   458→      setImporting(false);
   459→   459→    }
   460→   460→  };
   461→   461→
   462→   462→  const formatRupiah = (value: number) => {
   463→   463→    if (value >= 1_000_000_000_000) return `${(value / 1_000_000_000_000).toFixed(1)} T`;
   464→   464→    if (value >= 1_000_000_000) return `${(value / 1_000_000_000).toFixed(1)} M`;
   465→   465→    if (value >= 1_000_000) return `${(value / 1_000_000).toFixed(1)} Jt`;
   466→   466→    return value.toLocaleString("id-ID");
   467→   467→  };
   468→   468→
   469→   469→  return (
   470→   470→    <Dialog open={open} onOpenChange={handleClose}>
   471→   471→      <DialogContent className="sm:max-w-2xl max-h-[85vh]">
   472→   472→        <DialogHeader>
   473→   473→          <DialogTitle className="flex items-center gap-2">
   474→   474→            <div className={`w-2.5 h-2.5 rounded-full ${jenisColors[jenis]}`} />
   475→   475→            Import Data {jenisLabels[jenis]}
   476→   476→          </DialogTitle>
   477→   477→          <DialogDescription>
   478→   478→            Impor data {jenisLabels[jenis]} dari file Excel (.xlsx) atau CSV
   479→   479→          </DialogDescription>
   480→   480→        </DialogHeader>
   481→   481→
   482→   482→        <AnimatePresence mode="wait">
   483→   483→          {/* ====== STEP 1: UPLOAD ====== */}
   484→   484→          {step === "upload" && (
   485→   485→            <motion.div
   486→   486→              key="upload"
