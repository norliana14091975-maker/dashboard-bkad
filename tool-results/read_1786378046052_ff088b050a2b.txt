   144→function isValidHexColor(color: string): boolean {
   145→  return /^#[0-9A-Fa-f]{6}$/.test(color);

   148→function normalizeHex(value: string): string {
   149→  let hex = value.trim();
   150→  if (!hex.startsWith("#")) {
   151→    hex = "#" + hex;
   152→  }
   153→  return hex;
