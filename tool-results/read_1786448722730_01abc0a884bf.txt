   157→   157→// Extract prefixes from a list of kode akun
   158→   158→function extractPrefixesFromKodeAkun(kodeAkunList: string[]): Array<{ prefix: string; label: string; count: number }> {
   159→   159→  const prefixCount = new Map<string, number>();
   160→   160→
   161→   161→  for (const kode of kodeAkunList) {
   162→   162→    const segments = kode.split(".");
   163→   163→    if (segments.length < 2) continue;
   164→   164→
   165→   165→    // Level 1: "4", "5", "6"
   166→   166→    const l1 = segments[0];
   167→   167→    prefixCount.set(l1, (prefixCount.get(l1) || 0) + 1);
   168→   168→
   169→   169→    // Level 2: "4.1", "5.1", "5.2"
   170→   170→    if (segments.length >= 2) {
   171→   171→      const l2 = `${segments[0]}.${segments[1]}`;
   172→   172→      prefixCount.set(l2, (prefixCount.get(l2) || 0) + 1);
   173→   173→    }
   174→   174→
   175→   175→    // Level 3: "5.1.01", "5.2.02"
   176→   176→    if (segments.length >= 3) {
   177→   177→      const l3 = `${segments[0]}.${segments[1]}.${segments[2]}`;
   178→   178→      prefixCount.set(l3, (prefixCount.get(l3) || 0) + 1);
   179→   179→    }
   180→   180→  }
   181→   181→
   182→   182→  return Array.from(prefixCount.entries())
   183→   183→    .map(([prefix, count]) => ({
   184→   184→      prefix,
   185→   185→      label: KODE_AKUN_PREFIXES[prefix] || `Kode ${prefix}`,
   186→   186→      count,
   187→   187→    }))
   188→   188→    .sort((a, b) => a.prefix.localeCompare(b.prefix));
   189→   189→}
