    59→  const [loading, setLoading] = useState(true);
    60→  const [loaderExiting, setLoaderExiting] = useState(false); // For modern loader fade-out
    61→  const [error, setError] = useState<string | null>(null);
    62→  const [tahun, setTahun] = useState<number>(0); // 0 = not yet initialized
    63→  const [activeView, setActiveView] = useState<ActiveView>("dashboard");
    64→  const [sidebarOpen, setSidebarOpen] = useState(false);
    65→  const { stats: visitorStats } = useVisitorTracker(activeView);
    66→
    67→  // Setup wizard state
    68→  const [needsSetup, setNeedsSetup] = useState<boolean | null>(null); // null = checking

    70→  const MIN_LOADING_MS = pengaturan.loaderDisplayTime ?? 5000; // From settings
    71→  const minLoadingRef = useRef(MIN_LOADING_MS);
    72→  minLoadingRef.current = MIN_LOADING_MS;
    73→
    74→  // Auto-refresh state
    75→  const [nextRefreshIn, setNextRefreshIn] = useState<number>(0); // seconds until next refresh
    76→  const [isRefreshing, setIsRefreshing] = useState(false); // manual refresh spinner
    77→  const lastRefreshRef = useRef<number>(0); // 0 = not yet initialized

    81→
    82→  const fetchData = useCallback(async (tahunParam: number) => {
    83→    setLoading(true);
    84→    setLoaderExiting(false);
    85→    setError(null);
    86→    const startTime = Date.now();

    89→      const res = await fetch(`/api/dashboard?tahun=${tahunParam}`);
    90→      if (!res.ok) throw new Error("Failed to fetch dashboard data");
    91→      const json = await res.json();
    92→      setData(json);
    93→      // Auto-set tahun to the latest/active year on first load
    94→      if (tahun === 0 && json.tahunList?.length > 0) {
    95→        setTahun(json.tahun);
    96→      }
    97→    } catch (err) {
    98→      setError(err instanceof Error ? err.message : "Unknown error");
    99→    } finally {
   100→      // Ensure loader is visible for at least minDisplay ms (0 = no minimum)
   101→      if (minDisplay > 0) {
   102→        const elapsed = Date.now() - startTime;
   103→        const remaining = minDisplay - elapsed;
   104→        if (remaining > 0) {
   105→          await new Promise((r) => setTimeout(r, remaining));
   106→        }
   107→      }
   108→      // For modern loader: trigger fade-out animation before unmounting
   109→      if (pengaturan.loaderType === "modern") {
   110→        setLoaderExiting(true);
   111→      }
   112→      setLoading(false);
   113→      // Mark the initial data load time for auto-refresh countdown
   114→      lastRefreshRef.current = Date.now();
   115→    }
   116→  }, [tahun, pengaturan.loaderType]);
   117→
   118→  // Silent refresh — updates data in background without showing loader
   119→  const silentRefresh = useCallback(async (tahunParam: number) => {
   120→    try {
   121→      const res = await fetch(`/api/dashboard?tahun=${tahunParam}`);
   122→      if (!res.ok) throw new Error("Failed to fetch dashboard data");
   123→      const json = await res.json();
   124→      setData(json);
   125→      lastRefreshRef.current = Date.now();
   126→    } catch (err) {
   127→      // Silently fail — don't show error for background refreshes
   128→      console.warn("Auto-refresh failed:", err instanceof Error ? err.message : "Unknown error");
   129→    }
   130→  }, []);
   131→
   132→  // Check if setup is needed on mount
   133→  useEffect(() => {
   134→    fetch("/api/setup")
   135→      .then((res) => res.json())
   136→      .then((json) => {
   137→        setNeedsSetup(json.needsSetup === true);
   138→      })
   139→      .catch(() => {
   140→        // If error, assume setup is not needed (database exists)
   141→        setNeedsSetup(false);
   142→      });
   143→  }, []);
   144→
   145→  useEffect(() => {
   146→    // Don't fetch dashboard data if setup is needed
   147→    if (needsSetup === true || needsSetup === null) return;
   148→    // On first load, fetch the active year from the API
   149→    if (tahun === 0) {
   150→      fetch('/api/dashboard').then(res => res.json()).then(json => {
   151→        if (json.activeTahun) {
   152→          setTahun(json.activeTahun);
   153→        } else if (json.tahun) {
   154→          setTahun(json.tahun);
   155→        }
   156→      }).catch(() => {
   157→        setTahun(2024); // fallback
   158→      });
   159→    } else {
   160→      fetchData(tahun);
   161→    }
   162→  }, [tahun, fetchData, needsSetup]);
   163→
   164→  const handleSetupComplete = useCallback(() => {
   165→    setNeedsSetup(false);
   166→    // Reload to refresh all data
   167→    window.location.reload();
   168→  }, []);
   169→
   170→  // Helper: check if a view is hidden for the current user role
   171→  const isViewHidden = useCallback((viewId: string): boolean => {
   172→    const userRole = user?.role || "public";
   173→    const hiddenItems = pengaturan.sidebarConfig?.hiddenItems;
   174→    if (!hiddenItems) return false;
   175→    const roleHidden = hiddenItems[userRole] || [];
   176→    return roleHidden.includes(viewId);
   177→  }, [user?.role, pengaturan.sidebarConfig]);
   178→
   179→  // Listen for quick navigation events from DashboardView
   180→  useEffect(() => {
   181→    const handler = (e: Event) => {
   182→      const customEvent = e as CustomEvent<ActiveView>;
   183→      if (customEvent.detail && !isViewHidden(customEvent.detail)) {
   184→        setActiveView(customEvent.detail);
   185→      }
   186→    };
   187→    window.addEventListener('navigate-view', handler);
   188→    return () => window.removeEventListener('navigate-view', handler);
   189→  }, [isViewHidden]);
   190→
   191→  // Redirect to dashboard if activeView is hidden for current role
   192→  useEffect(() => {
   193→    if (activeView !== "dashboard" && isViewHidden(activeView)) {
   194→      setActiveView("dashboard");
   195→    }
   196→  }, [activeView, isViewHidden]);
   197→
   198→  // Auto-refresh interval logic — robust implementation
   199→  // Only depends on autoRefreshInterval and tahun. Does NOT depend on `loading`
   200→  // to prevent the timer from resetting every time loading state changes.
   201→  useEffect(() => {
   202→    // Clear any existing refresh timer
   203→    if (autoRefreshTimerRef.current) {
   204→      clearTimeout(autoRefreshTimerRef.current);
   205→      autoRefreshTimerRef.current = null;
   206→    }
   207→
   208→    if (autoRefreshInterval <= 0 || tahun === 0) {
   209→      setNextRefreshIn(0);
   210→      lastRefreshRef.current = 0;
   211→      return;
   212→    }
   213→
   214→    const intervalMs = autoRefreshInterval * 60 * 1000; // Convert minutes to ms
   215→
   216→    // Only initialize lastRefreshRef if it hasn't been set yet
   217→    // (i.e., first time auto-refresh activates, or after it was reset)
   218→    if (lastRefreshRef.current === 0) {
   219→      lastRefreshRef.current = Date.now();
   220→    }
   221→
   222→    // Countdown timer — updates every second, reads from ref (no re-render dependency)
   223→    const countdownInterval = setInterval(() => {
   224→      const elapsed = Date.now() - lastRefreshRef.current;
   225→      const remaining = Math.max(0, Math.ceil((intervalMs - elapsed) / 1000));
   226→      setNextRefreshIn(remaining);
   227→    }, 1000);
   228→
   229→    // Auto-refresh using setTimeout (schedules next refresh after current one completes)
   230→    // This prevents interval drift and overlapping refreshes
   231→    const scheduleRefresh = () => {
   232→      const elapsed = Date.now() - lastRefreshRef.current;
   233→      const delay = Math.max(0, intervalMs - elapsed);
   234→      autoRefreshTimerRef.current = setTimeout(async () => {
   235→        await silentRefresh(tahun);
   236→        // After refresh completes, update the ref and schedule the next one
   237→        // (silentRefresh already sets lastRefreshRef.current = Date.now())
   238→        // Only reschedule if the interval setting hasn't changed
   239→        if (autoRefreshIntervalRef.current === autoRefreshInterval) {
   240→          scheduleRefresh();
   241→        }
   242→      }, delay);
   243→    };
   244→    scheduleRefresh();
   245→
   246→    return () => {
   247→      clearInterval(countdownInterval);
   248→      if (autoRefreshTimerRef.current) {
   249→        clearTimeout(autoRefreshTimerRef.current);
   250→        autoRefreshTimerRef.current = null;
   251→      }
   252→    };
   253→  }, [autoRefreshInterval, tahun, silentRefresh]);
   254→
   255→  // Show setup wizard if needed (after all hooks)
   256→  if (needsSetup === null) {
   257→    return (
   258→      <div className="min-h-screen flex items-center justify-center bg-gradient-to-br from-emerald-50 via-white to-amber-50">
   259→        <div className="flex flex-col items-center gap-4">
   260→          <div className="animate-spin w-10 h-10 border-4 border-emerald-600 border-t-transparent rounded-full" />
   261→          <p className="text-sm text-muted-foreground">Memeriksa konfigurasi...</p>
   262→        </div>
   263→      </div>
   264→    );
   265→  }
   266→
   267→  if (needsSetup === true) {
   268→    return <SetupWizard onComplete={handleSetupComplete} />;
   269→  }
   270→
   271→  const handleTahunChange = (newTahun: number) => {
   272→    // Reset the auto-refresh countdown when user changes tahun
   273→    lastRefreshRef.current = 0; // Will be re-initialized when fetchData completes
   274→    setTahun(newTahun);
   275→  };
   276→
   277→  const handleViewChange = (view: ActiveView) => {
   278→    setActiveView(view);
   279→  };
   280→
   281→  // Helper to get tahun list as numbers for backward compatibility
   282→  const tahunListNumbers = data?.tahunList?.map(t => t.tahun) || [];
   283→  const activeTahunFromData = data?.tahunList?.find(t => t.aktif)?.tahun || tahun;
   284→
   285→  const renderContent = () => {
   286→    // Admin/OPD view: show login if not authenticated, panel if authenticated
   287→    if (activeView === "admin") {
   288→      if (authLoading) {
   289→        return (
   290→          <div className="flex items-center justify-center py-20">
   291→            <div className="animate-spin w-8 h-8 border-4 border-primary border-t-transparent rounded-full" />
   292→          </div>
   293→        );
   294→      }
   295→      if (!isAuthenticated) {
   296→        return <LoginForm />;
   297→      }
   298→      // Show OPD Panel for OPD role, Admin Panel for admin/superadmin/bupati
   299→      if (user?.role === "opd") {
   300→        return <OpdPanel tahun={tahun} tahunList={data?.tahunList || []} />;
   301→      }
   302→      return <AdminPanel tahun={tahun} tahunList={data?.tahunList || []} />;
   303→    }
   304→
   305→    // Modern loader: keep showing during fade-out animation
   306→    if (loading || loaderExiting) {
   307→      if (pengaturan.loaderType === "modern") {
   308→        return <ModernSplashLoader isLoading={loading} onExitComplete={() => setLoaderExiting(false)} />;
   309→      }
   310→      if (loading) return <LoadingSkeleton />;
   311→    }
   312→    if (error) return <ErrorState error={error} onRetry={() => fetchData(tahun)} />;
   313→    if (!data) return null;
   314→
   315→    switch (activeView) {
   316→      case "dashboard":
   317→        return <DashboardView data={data} />;
   318→      case "ringkasan-eksekutif":
   319→        return <ExecutiveSummaryView data={data} />;
   320→      case "analisis-risiko":
   321→        return <AnalisisRisikoView data={data} />;
   322→      case "copilot":
   323→        return <FinancialCopilotView data={data} />;
   324→      case "apbd":
   325→        return <APBDTable data={data} />;
   326→      case "pendapatan":
   327→        return <PendapatanView data={data} />;
   328→      case "belanja":
   329→        return <BelanjaView data={data} />;
   330→      case "pembiayaan":
   331→        return <PembiayaanView data={data} />;
   332→      case "realisasi-akun":
   333→        return <RealisasiAkunView data={data} />;
   334→      case "realisasi-skpd":
   335→        return <RealisasiSkpdView data={data} />;
   336→      case "opd":
   337→        return <OpdView data={data} />;
   338→      case "transparansi":
   339→        return <TransparansiView data={data} />;
   340→      default:
   341→        return <DashboardView data={data} />;
   342→    }
   343→  };
   344→
   345→  // When showing login form for admin, don't show the sidebar/header layout
   346→  if (activeView === "admin" && !authLoading && !isAuthenticated) {
   347→    return (
   348→      <div className="min-h-screen bg-background">
   349→        <LoginForm />
   350→      </div>
   351→    );
   352→  }
   353→
   354→  return (
   355→    <div className="min-h-screen flex bg-background">
   356→      <Sidebar
   357→        activeView={activeView}
   358→        onViewChange={handleViewChange}
   359→        isOpen={sidebarOpen}
   360→        onToggle={() => setSidebarOpen(!sidebarOpen)}
   361→      />
   362→
   363→      <div className="flex-1 flex flex-col min-w-0 lg:ml-[60px]">
   364→        <DashboardHeader
   365→          activeView={activeView}
   366→          tahun={tahun}
   367→          tahunList={data?.tahunList || []}
   368→          activeTahun={activeTahunFromData}
   369→          onTahunChange={handleTahunChange}
   370→          onMenuToggle={() => setSidebarOpen(!sidebarOpen)}
   371→          onNavigateDashboard={() => setActiveView("dashboard")}
   372→          autoRefreshInterval={autoRefreshInterval}
   373→          nextRefreshIn={nextRefreshIn}
   374→          onManualRefresh={() => {
   375→            if (tahun > 0) {
   376→              setIsRefreshing(true);
   377→              // Reset the auto-refresh countdown when user manually refreshes
   378→              lastRefreshRef.current = Date.now();
   379→              silentRefresh(tahun).finally(() => setIsRefreshing(false));
   380→            }
   381→          }}
   382→          isRefreshing={isRefreshing}
   383→          visitorOnline={visitorStats.online}
   384→          visitorToday={visitorStats.today}
   385→        />
   386→
   387→        <main className="flex-1 p-4 pb-20 lg:p-6 lg:pb-6 overflow-auto">
   388→          {/* Auto-refresh inline indicator (compact, moved to header as primary) */}
   389→          {autoRefreshInterval > 0 && !loading && nextRefreshIn > 0 && data && (
   390→            <div className="flex items-center justify-end mb-2 lg:hidden">
   391→              <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground bg-muted/60 rounded-full px-2.5 py-1 backdrop-blur-sm">
   392→                <RefreshCw className="w-3 h-3 opacity-60" />
   393→                <span>
   394→                  Refresh dalam {nextRefreshIn >= 60 ? `${Math.floor(nextRefreshIn / 60)}m ${nextRefreshIn % 60}s` : `${nextRefreshIn}s`}
   395→                </span>
   396→              </div>
   397→            </div>
   398→          )}
   399→          <AnimatePresence mode="wait">
   400→            <motion.div
   401→              key={activeView}
   402→              initial={{ opacity: 0, y: 12 }}
   403→              animate={{ opacity: 1, y: 0 }}
   404→              exit={{ opacity: 0, y: -12 }}
   405→              transition={{ duration: 0.35, ease: "easeInOut" }}
   406→            >
   407→              {renderContent()}
   408→            </motion.div>
   409→          </AnimatePresence>
   410→        </main>
   411→
   412→        {/* Footer — hidden on mobile (replaced by MobileBottomNav) */}
   413→        <footer className="hidden lg:block bg-white/60 dark:bg-slate-900/60 backdrop-blur-sm border-t border-black/[0.06] dark:border-white/[0.06] px-4 lg:px-6 py-3 mt-auto">
   414→          <div className="flex items-center justify-between">
   415→            <div className="flex items-center gap-2">
   416→              <img
   417→                src={logoSrc}
   418→                alt="Logo Kabupaten Seruyan"
   419→                width={16}
   420→                height={16}
   421→                className="w-4 h-4 object-contain"
   422→              />
   423→              <span className="text-xs text-muted-foreground">
   424→                © {new Date().getFullYear()} {pengaturan.namaPemerintah}
   425→              </span>
   426→              <span className="inline-flex items-center rounded-full bg-black/[0.06] dark:bg-white/[0.08] px-1.5 py-0.5 text-[10px] font-mono font-semibold text-muted-foreground">
   427→                v{pengaturan.versiAplikasi || "1.0.0"}
   428→              </span>
   429→            </div>
   430→            <div className="flex items-center gap-1.5 text-xs text-muted-foreground">
   431→              <span className="relative flex h-1.5 w-1.5">
   432→                <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-green-500 opacity-75" />
   433→                <span className="relative inline-flex rounded-full h-1.5 w-1.5 bg-green-500" />
   434→              </span>
   435→              <span className="font-semibold">{visitorStats.online}</span> online
   436→              <span className="text-muted-foreground/40">·</span>
   437→              <span><span className="font-semibold">{visitorStats.today}</span> today</span>
   438→              <span className="text-muted-foreground/40">·</span>
   439→              <span><span className="font-semibold">{visitorStats.total}</span> total</span>
   440→            </div>
   441→          </div>
   442→        </footer>
   443→      </div>
   444→
   445→      {/* Mobile Bottom Navigation — shown only on mobile */}
   446→      <MobileBottomNav
   447→        activeView={activeView}
   448→        onViewChange={handleViewChange}
   449→        onOpenSidebar={() => setSidebarOpen(true)}
   450→      />
   451→    </div>
   452→  );
   453→}

   465→// ============ DASHBOARD VIEW (MODERN ANIMATED) ============
   466→function DashboardView({ data }: { data: DashboardData }) {
   467→  const { pengaturan, logoSrc } = usePengaturan();
   468→  const { user } = useAuth();
   469→  const containerVariants = {
   470→    hidden: { opacity: 0 },
   471→    visible: {
   472→      opacity: 1,
   473→      transition: { staggerChildren: 0.08, delayChildren: 0.1 },
   474→    },
   475→  };
   476→
   477→  const itemVariants = {
   478→    hidden: { opacity: 0, y: 30, scale: 0.97 },
   479→    visible: {
   480→      opacity: 1,
   481→      y: 0,
   482→      scale: 1,
   483→      transition: { duration: 0.6, ease: [0.22, 1, 0.36, 1] },
   484→    },
   485→  };
   486→
   487→  // Helper: check if a view is hidden for the current user role
   488→  const isViewHiddenForUser = (viewId: string): boolean => {
   489→    const userRole = user?.role || "public";
   490→    const hiddenItems = pengaturan.sidebarConfig?.hiddenItems;
   491→    if (!hiddenItems) return false;
   492→    const roleHidden = hiddenItems[userRole] || [];
   493→    return roleHidden.includes(viewId);
   494→  };
   495→
   496→  // Quick navigation items — filtered by sidebar visibility and copilot config
   497→  const copilotEnabled = pengaturan.copilotConfig?.enabled ?? true;
   498→  const allQuickNavItems = [
   499→    { id: "ringkasan-eksekutif" as ActiveView, label: "Ringkasan Eksekutif", icon: BarChart3, color: "from-violet-500 to-purple-600", desc: "Executive Summary" },
   500→    { id: "analisis-risiko" as ActiveView, label: "Analisis Risiko", icon: AlertTriangle, color: "from-rose-500 to-red-600", desc: "Risk Analysis" },
   501→    { id: "copilot" as ActiveView, label: "AI Copilot", icon: BotMessageSquare, color: "from-amber-500 to-yellow-600", desc: "Tanya AI Keuangan" },
   502→    { id: "realisasi-skpd" as ActiveView, label: "Realisasi SKPD", icon: TrendingUp, color: "from-teal-500 to-cyan-600", desc: "Per-SKPD/OPD" },
   503→  ];
   504→
   505→  const quickNavItems = allQuickNavItems.filter((item) => {
   506→    if (isViewHiddenForUser(item.id)) return false;
   507→    if (item.id === "copilot" && !copilotEnabled) return false;
   508→    return true;
   509→  });
   510→
   511→  return (
   512→    <motion.div
   513→      variants={containerVariants}
   514→      initial="hidden"
   515→      animate="visible"

  1125→function ModernSplashLoader({ isLoading, onExitComplete }: { isLoading: boolean; onExitComplete: () => void }) {
  1126→  const { pengaturan, logoSrc } = usePengaturan();
  1127→  const MIN_LOADING_MS = pengaturan.loaderDisplayTime ?? 5000;
  1128→
  1129→  // When loading completes, give the SplashLoader time to animate its exit
  1130→  // before the parent unmounts it
  1131→  useEffect(() => {
  1132→    if (!isLoading) {
  1133→      const timer = setTimeout(() => {
  1134→        onExitComplete();
  1135→      }, 1200);
  1136→      return () => clearTimeout(timer);
  1137→    }
  1138→  }, [isLoading, onExitComplete]);
  1139→
  1140→  return (
  1141→    <SplashLoader
  1142→      logoSrc={logoSrc}
  1143→      namaPemerintah={pengaturan.namaPemerintah}
  1144→      namaInstansi={pengaturan.namaInstansi}
  1145→      warnaPrimary={pengaturan.warnaPrimary}
  1146→      warnaSecondary={pengaturan.warnaSecondary}
  1147→      warnaAccent={pengaturan.warnaAccent}
  1148→      warnaDark={pengaturan.warnaDark}
  1149→      waktuTampil={MIN_LOADING_MS}
  1150→      isLoading={isLoading}
  1151→    />
  1152→  );
  1153→}
