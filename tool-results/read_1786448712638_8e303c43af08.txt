   181→
   182→  /**
   183→   * Split a CSV line respecting quoted fields.
   184→   * Supports both comma and semicolon as delimiter (auto-detect).
   185→   * Quoted fields (e.g., "31.500,5") are preserved as single values.
   186→   */
   187→  const splitCSVLine = (line: string, delimiter: string): string[] => {
   188→    const result: string[] = [];
   189→    let current = '';
   190→    let inQuotes = false;
   191→
   192→    for (let i = 0; i < line.length; i++) {
   193→      const ch = line[i];
   194→      if (ch === '"') {
   195→        if (inQuotes && i + 1 < line.length && line[i + 1] === '"') {
   196→          // Escaped quote
   197→          current += '"';
   198→          i++;
   199→        } else {
   200→          inQuotes = !inQuotes;
   201→        }
   202→      } else if (ch === delimiter && !inQuotes) {
   203→        result.push(current.trim());
   204→        current = '';
   205→      } else {
   206→        current += ch;
   207→      }
   208→    }
   209→    result.push(current.trim());
   210→    return result;
   211→  };
   212→
   213→  const parseCSV = (text: string): { rows: ImportRow[]; errors: string[] } => {
   214→    const lines = text
   215→      .split("\n")
   216→      .map((l) => l.trim())
   217→      .filter((l) => l && !l.startsWith("#"));
   218→
   219→    if (lines.length === 0) {
   220→      return { rows: [], errors: ["File kosong"] };
   221→    }
   222→
   223→    // Auto-detect delimiter: if first data line has semicolons, use semicolon
   224→    // This is common in Indonesian/European CSV where comma is the decimal separator
   225→    const firstDataLine = lines.find(l => !l.startsWith('#')) || '';
   226→    const delimiter = firstDataLine.includes(';') ? ';' : ',';
   227→
   228→    // Skip header line
   229→    const dataLines = lines.slice(1);
   230→    const rows: ImportRow[] = [];
   231→    const errors: string[] = [];
   232→
   233→    dataLines.forEach((line, idx) => {
   234→      const cols = splitCSVLine(line, delimiter);
   235→      if (cols.length < 5) {
   236→        errors.push(`Baris ${idx + 2}: Kolom tidak lengkap (butuh 5 kolom)`);
   237→        return;
   238→      }
   239→
   240→      const [kodeAkun, namaAkun, kategori, anggaranStr, realisasiStr] = cols;
   241→
   242→      if (!kodeAkun) {
   243→        errors.push(`Baris ${idx + 2}: Kode Akun kosong`);
   244→        return;
   245→      }
   246→      if (!namaAkun) {
   247→        errors.push(`Baris ${idx + 2}: Nama Akun kosong`);
   248→        return;
   249→      }
   250→      if (!kategori) {
   251→        errors.push(`Baris ${idx + 2}: Kategori kosong`);
   252→        return;
   253→      }
   254→
   255→      const anggaran = parseNumber(anggaranStr);
   256→      const realisasi = parseNumber(realisasiStr);
   257→
   258→      if (isNaN(anggaran) || anggaran < 0) {
   259→        errors.push(`Baris ${idx + 2}: Anggaran tidak valid ("${anggaranStr}")`);
   260→        return;
   261→      }
   262→      if (isNaN(realisasi) || realisasi < 0) {
   263→        errors.push(`Baris ${idx + 2}: Realisasi tidak valid ("${realisasiStr}")`);
   264→        return;
   265→      }
   266→
   267→      rows.push({ kodeAkun, namaAkun, kategori, anggaran, realisasi });
   268→    });
   269→
   270→    return { rows, errors };
   271→  };
   272→
   273→  const parseXLSX = (data: ArrayBuffer): { rows: ImportRow[]; errors: string[] } => {
   274→    try {
   275→      const workbook = XLSX.read(data, { type: 'array' });
   276→
   277→      // Try to find the "Template Data" sheet first, otherwise use first sheet
   278→      let sheetName = workbook.SheetNames.find(name =>
   279→        name.toLowerCase().includes('template') || name.toLowerCase().includes('data')
   280→      );
   281→      if (!sheetName) {
   282→        sheetName = workbook.SheetNames[0];
   283→      }
   284→
   285→      const ws = workbook.Sheets[sheetName];
   286→      if (!ws) {
   287→        return { rows: [], errors: ["Sheet tidak ditemukan dalam file"] };
   288→      }
   289→
   290→      // Convert sheet to array of arrays
   291→      const sheetData: unknown[][] = XLSX.utils.sheet_to_json(ws, { header: 1, defval: '' });
   292→
   293→      if (sheetData.length === 0) {
   294→        return { rows: [], errors: ["File kosong"] };
   295→      }
   296→
   297→      // Skip header row (first row)
   298→      const dataRows = sheetData.slice(1);
   299→      const rows: ImportRow[] = [];
   300→      const errors: string[] = [];
   301→
   302→      dataRows.forEach((row, idx) => {
   303→        // Skip empty rows (but keep rows where anggaran/realisasi is 0)
   304→        if (!row || row.length === 0 || row.every(cell => cell == null || cell === '')) {
   305→          return;
   306→        }
   307→
   308→        const kodeAkun = String(row[0] ?? '').trim();
   309→        const namaAkun = String(row[1] ?? '').trim();
   310→        const kategori = String(row[2] ?? '').trim();
   311→        // Use ?? instead of || to preserve 0 values (0 || '' evaluates to '')
   312→        const anggaranStr = String(row[3] ?? '').trim();
   313→        const realisasiStr = String(row[4] ?? '').trim();
   314→
   315→        // Skip rows with no kode akun (likely empty template rows)
   316→        if (!kodeAkun) return;
   317→
   318→        if (!namaAkun) {
   319→          errors.push(`Baris ${idx + 2}: Nama Akun kosong`);
   320→          return;
   321→        }
   322→        if (!kategori) {
   323→          errors.push(`Baris ${idx + 2}: Kategori kosong`);
   324→          return;
   325→        }
   326→
   327→        const anggaran = parseNumber(anggaranStr);
   328→        const realisasi = parseNumber(realisasiStr);
   329→
   330→        if (isNaN(anggaran) || anggaran < 0) {
   331→          errors.push(`Baris ${idx + 2}: Anggaran tidak valid ("${anggaranStr}")`);
   332→          return;
   333→        }
   334→        if (isNaN(realisasi) || realisasi < 0) {
   335→          errors.push(`Baris ${idx + 2}: Realisasi tidak valid ("${realisasiStr}")`);
   336→          return;
   337→        }
   338→
   339→        rows.push({ kodeAkun, namaAkun, kategori, anggaran, realisasi });
   340→      });
   341→
   342→      return { rows, errors };
   343→    } catch (err) {
   344→      return { rows: [], errors: [`Gagal membaca file XLSX: ${err instanceof Error ? err.message : 'Unknown error'}`] };
   345→    }
   346→  };
   347→
   348→  const parseNumber = (str: string): number => {
   349→    if (!str || !str.trim()) return NaN;
   350→    // Remove "Rp" / "Rp." prefix and whitespace
   351→    let cleaned = str.replace(/Rp\.?\s*/gi, '').trim();
   352→    // Remove dots as thousand separators (Indonesian format: 31.500.000.000)
   353→    cleaned = cleaned.replace(/\./g, '');
   354→    // Replace comma with dot as decimal separator (Indonesian format: 31500,5 → 31500.5)
   355→    cleaned = cleaned.replace(/,/g, '.');
   356→    const num = parseFloat(cleaned);
   357→    return num;
   358→  };
   359→
   360→  // Auto-detect jenis and kategori from kode akun prefix
   361→  const enrichRowsWithAutoDetect = (rows: ImportRow[]): ImportRow[] => {
   362→    if (!isAutoDetect) return rows;
   363→    return rows.map(row => {
   364→      const firstDigit = row.kodeAkun.trim().charAt(0);
   365→      let detectedJenis: JenisData | null = null;
   366→      let detectedKategori = row.kategori; // keep existing kategori from file
   367→
   368→      switch (firstDigit) {
   369→        case '4':
   370→          detectedJenis = 'pendapatan';
   371→          if (!row.kategori) {
   372→            const seg2 = row.kodeAkun.split('.')[1];
   373→            detectedKategori = seg2 === '1' ? 'PAD' : seg2 === '2' ? 'Transfer' : seg2 === '3' ? 'Lain-Lain' : 'PAD';
   374→          }
   375→          break;
   376→        case '5':
   377→          detectedJenis = 'belanja';
   378→          if (!row.kategori) {
   379→            const seg2 = row.kodeAkun.split('.')[1];
   380→            detectedKategori = seg2 === '1' ? 'Operasi' : seg2 === '2' ? 'Modal' : seg2 === '3' ? 'Tidak Terduga' : seg2 === '4' ? 'Transfer' : 'Operasi';
   381→          }
   382→          break;
   383→        case '6':
   384→          detectedJenis = 'pembiayaan';
   385→          if (!row.kategori) {
   386→            const seg2 = row.kodeAkun.split('.')[1];
   387→            detectedKategori = seg2 === '1' ? 'Penerimaan' : seg2 === '2' ? 'Pengeluaran' : 'Penerimaan';
   388→          }
   389→          break;
   390→      }
   391→
   392→      return {
   393→        ...row,
   394→        kategori: detectedKategori,
   395→        jenis: detectedJenis || undefined,
   396→        detectedJenis,
   397→        detectedKategori,
   398→      };
   399→    });
   400→  };
   401→
   402→  const handleFile = (file: File) => {
   403→    const isXLSX = file.name.endsWith('.xlsx') || file.name.endsWith('.xls');
   404→    const isCSV = file.name.endsWith('.csv') || file.name.endsWith('.txt');
   405→
   406→    if (!isXLSX && !isCSV) {
   407→      toast({
   408→        title: "Format Salah",
   409→        description: "Hanya file XLSX (Excel) dan CSV yang didukung",
   410→        variant: "destructive",
   411→      });
   412→      return;
   413→    }
   414→
   415→    setUploadedFileName(file.name);
   416→
   417→    if (isXLSX) {
   418→      const reader = new FileReader();
   419→      reader.onload = (e) => {
   420→        const data = e.target?.result as ArrayBuffer;
   421→        const { rows, errors } = parseXLSX(data);
   422→        setParseErrors(errors);
   423→        setParsedRows(enrichRowsWithAutoDetect(rows));
   424→
   425→        if (rows.length > 0) {
   426→          setStep("preview");
   427→        } else if (errors.length > 0) {
   428→          toast({
   429→            title: "Gagal Membaca File",
   430→            description: "Tidak ada data valid yang dapat diimpor",
   431→            variant: "destructive",
   432→          });
   433→        } else {
   434→          toast({
   435→            title: "File Kosong",
   436→            description: "Tidak ada data ditemukan dalam file Excel",
   437→            variant: "destructive",
   438→          });
   439→        }
   440→      };
   441→      reader.readAsArrayBuffer(file);
   442→    } else {
   443→      // CSV
   444→      const reader = new FileReader();
   445→      reader.onload = (e) => {
   446→        const text = e.target?.result as string;
   447→        const { rows, errors } = parseCSV(text);
   448→        setParseErrors(errors);
   449→        setParsedRows(enrichRowsWithAutoDetect(rows));
   450→
   451→        if (rows.length > 0) {
   452→          setStep("preview");
   453→        } else if (errors.length > 0) {
   454→          toast({
   455→            title: "Gagal Membaca File",
   456→            description: "Tidak ada data valid yang dapat diimpor",
   457→            variant: "destructive",
   458→          });
   459→        }
   460→      };
   461→      reader.readAsText(file);
   462→    }
   463→  };
   464→
   465→  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   466→    const file = e.target.files?.[0];
   467→    if (file) handleFile(file);
   468→    // Reset the input so the same file can be re-selected
   469→    e.target.value = "";
   470→  };
   471→
   472→  const handleDrop = (e: React.DragEvent) => {
   473→    e.preventDefault();
   474→    setDragOver(false);
   475→    const file = e.dataTransfer.files?.[0];
   476→    if (file) handleFile(file);
   477→  };
   478→
   479→  const handleDragOver = (e: React.DragEvent) => {
   480→    e.preventDefault();
   481→    setDragOver(true);
   482→  };
   483→
   484→  const handleDragLeave = () => {
   485→    setDragOver(false);
   486→  };
   487→
   488→  const removeRow = (index: number) => {
   489→    setParsedRows((prev) => prev.filter((_, i) => i !== index));
   490→  };
   491→
   492→  const handleImport = async () => {
   493→    if (!tahunAnggaranId || parsedRows.length === 0) return;
   494→
   495→    setImporting(true);
   496→    try {
   497→      const res = await fetch("/api/admin/import", {
   498→        method: "POST",
   499→        headers: { "Content-Type": "application/json" },
   500→        credentials: "include",
   501→        body: JSON.stringify({
   502→          jenis: isAutoDetect ? "auto" : effectiveJenis,
   503→          tahunAnggaranId,
   504→          rows: parsedRows,
   505→          mode,
   506→          opdId: selectedOpdId && selectedOpdId !== "__none__" ? selectedOpdId : null,
   507→        }),
   508→      });
   509→
   510→      const json = await res.json();
   511→
   512→      if (!res.ok) {
   513→        if (json.validationErrors) {
   514→          const errMsg = json.validationErrors
   515→            .slice(0, 5)
   516→            .map((v: { row: number; error: string }) => `Baris ${v.row}: ${v.error}`)
   517→            .join("\n");
   518→          throw new Error(`Validasi gagal:\n${errMsg}`);
   519→        }
   520→        throw new Error(json.error || "Gagal mengimpor data");
   521→      }
   522→
   523→      setResult({
   524→        success: true,
   525→        imported: json.imported,
   526→        created: json.created || 0,
   527→        updated: json.updated || 0,
   528→        message: json.message,
   529→      });
   530→      setStep("result");
   531→      onSuccess();
   532→    } catch (err) {
   533→      toast({
   534→        title: "Gagal Import",
   535→        description: err instanceof Error ? err.message : "Terjadi kesalahan",
   536→        variant: "destructive",
   537→      });
   538→      setResult({
   539→        success: false,
   540→        imported: 0,
   541→        created: 0,
   542→        updated: 0,
   543→        message: err instanceof Error ? err.message : "Terjadi kesalahan",
   544→      });
   545→      setStep("result");
   546→    } finally {
   547→      setImporting(false);
   548→    }
   549→  };
   550→
   551→  const formatRupiah = (value: number) => {
   552→    if (value >= 1_000_000_000_000) return `${(value / 1_000_000_000_000).toFixed(1)} T`;
   553→    if (value >= 1_000_000_000) return `${(value / 1_000_000_000).toFixed(1)} M`;
   554→    if (value >= 1_000_000) return `${(value / 1_000_000).toFixed(1)} Jt`;
   555→    return value.toLocaleString("id-ID");
   556→  };
   557→
   558→  return (
   559→    <Dialog open={open} onOpenChange={handleClose}>
   560→      <DialogContent className="sm:max-w-2xl max-h-[85vh]">
   561→        <DialogHeader>
   562→          <DialogTitle className="flex items-center gap-2">
   563→            <div className={`w-2.5 h-2.5 rounded-full ${effectiveJenis ? jenisColors[effectiveJenis] : 'bg-sky-500'}`} />
   564→            Import Data{isAutoDetect ? " — Auto-Detect" : ` ${jenisLabels[effectiveJenis!]}`}
   565→          </DialogTitle>
   566→          <DialogDescription>
   567→            {isAutoDetect
