   125→      lastRefreshRef.current = Date.now();

   133→  useEffect(() => {

   199→  // Only depends on autoRefreshInterval and tahun. Does NOT depend on `loading`
   200→  // to prevent the timer from resetting every time loading state changes.
   201→  useEffect(() => {
   202→    // Clear any existing refresh timer
   203→    if (autoRefreshTimerRef.current) {
   204→      clearTimeout(autoRefreshTimerRef.current);
   205→      autoRefreshTimerRef.current = null;
   206→    }
   207→
   208→    if (autoRefreshInterval <= 0 || tahun === 0) {
   209→      setNextRefreshIn(0);
   210→      lastRefreshRef.current = 0;
   211→      return;
   212→    }
   213→
   214→    const intervalMs = autoRefreshInterval * 60 * 1000; // Convert minutes to ms
   215→
   216→    // Only initialize lastRefreshRef if it hasn't been set yet
   217→    // (i.e., first time auto-refresh activates, or after it was reset)
   218→    if (lastRefreshRef.current === 0) {
   219→      lastRefreshRef.current = Date.now();
   220→    }
   221→
   222→    // Countdown timer — updates every second, reads from ref (no re-render dependency)
   223→    const countdownInterval = setInterval(() => {
   224→      const elapsed = Date.now() - lastRefreshRef.current;
   225→      const remaining = Math.max(0, Math.ceil((intervalMs - elapsed) / 1000));
   226→      setNextRefreshIn(remaining);
   227→    }, 1000);
   228→
   229→    // Auto-refresh using setTimeout (schedules next refresh after current one completes)
   230→    // This prevents interval drift and overlapping refreshes
   231→    const scheduleRefresh = () => {
   232→      const elapsed = Date.now() - lastRefreshRef.current;
   233→      const delay = Math.max(0, intervalMs - elapsed);
   234→      autoRefreshTimerRef.current = setTimeout(async () => {
   235→        await silentRefresh(tahun);
   236→        // After refresh completes, update the ref and schedule the next one
   237→        // (silentRefresh already sets lastRefreshRef.current = Date.now())
   238→        // Only reschedule if the interval setting hasn't changed
   239→        if (autoRefreshIntervalRef.current === autoRefreshInterval) {
   240→          scheduleRefresh();
   241→        }
   242→      }, delay);
   243→    };
   244→    scheduleRefresh();
   245→
   246→    return () => {
   247→      clearInterval(countdownInterval);
   248→      if (autoRefreshTimerRef.current) {
   249→        clearTimeout(autoRefreshTimerRef.current);
   250→        autoRefreshTimerRef.current = null;
   251→      }
   252→    };
   253→  }, [autoRefreshInterval, tahun, silentRefresh]);

   255→  // Show setup wizard if needed (after all hooks)
   256→  if (needsSetup === null) {

   275→  };
   276→
   277→  const handleViewChange = (view: ActiveView) => {

   296→        return <LoginForm />;
   297→      }
   298→      // Show OPD Panel for OPD role, Admin Panel for admin/superadmin/bupati
   299→      if (user?.role === "opd") {
   300→        return <OpdPanel tahun={tahun} tahunList={data?.tahunList || []} />;
   301→      }
   302→      return <AdminPanel tahun={tahun} tahunList={data?.tahunList || []} />;
   303→    }
   304→
   305→    // Modern loader: keep showing during fade-out animation
   306→    if (loading || loaderExiting) {
   307→      if (pengaturan.loaderType === "modern") {
   308→        return <ModernSplashLoader isLoading={loading} onExitComplete={() => setLoaderExiting(false)} />;
   309→      }
   310→      if (loading) return <LoadingSkeleton />;
   311→    }
   312→    if (error) return <ErrorState error={error} onRetry={() => fetchData(tahun)} />;
   313→    if (!data) return null;
   314→
   315→    switch (activeView) {
   316→      case "dashboard":
   317→        return <DashboardView data={data} />;
   318→      case "ringkasan-eksekutif":
   319→        return <ExecutiveSummaryView data={data} />;
   320→      case "analisis-risiko":
   321→        return <AnalisisRisikoView data={data} />;
   322→      case "copilot":
   323→        return <FinancialCopilotView data={data} />;
   324→      case "apbd":
   325→        return <APBDTable data={data} />;
   326→      case "pendapatan":
   327→        return <PendapatanView data={data} />;
   328→      case "belanja":
   329→        return <BelanjaView data={data} />;
   330→      case "pembiayaan":
   331→        return <PembiayaanView data={data} />;
   332→      case "realisasi-akun":
   333→        return <RealisasiAkunView data={data} />;
   334→      case "realisasi-skpd":
   335→        return <RealisasiSkpdView data={data} />;
   336→      case "opd":
   337→        return <OpdView data={data} />;
   338→      case "transparansi":
   339→        return <TransparansiView data={data} />;
   340→      default:

   364→        <DashboardHeader
   365→          activeView={activeView}
   366→          tahun={tahun}
   367→          tahunList={data?.tahunList || []}
   368→          activeTahun={activeTahunFromData}
   369→          onTahunChange={handleTahunChange}
   370→          onMenuToggle={() => setSidebarOpen(!sidebarOpen)}
   371→          onNavigateDashboard={() => setActiveView("dashboard")}
   372→          autoRefreshInterval={autoRefreshInterval}
   373→          nextRefreshIn={nextRefreshIn}
   374→          onManualRefresh={() => {
   375→            if (tahun > 0) {
   376→              setIsRefreshing(true);
   377→              // Reset the auto-refresh countdown when user manually refreshes
   378→              lastRefreshRef.current = Date.now();
   379→              silentRefresh(tahun).finally(() => setIsRefreshing(false));
   380→            }
   381→          }}
   382→          isRefreshing={isRefreshing}
   383→          visitorOnline={visitorStats.online}
   384→          visitorToday={visitorStats.today}
   385→        />
   386→
   387→        <main className="flex-1 p-4 pb-20 lg:p-6 lg:pb-6 overflow-auto">
   388→          {/* Auto-refresh inline indicator (compact, moved to header as primary) */}
   389→          {autoRefreshInterval > 0 && !loading && nextRefreshIn > 0 && data && (
   390→            <div className="flex items-center justify-end mb-2 lg:hidden">
   391→              <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground bg-muted/60 rounded-full px-2.5 py-1 backdrop-blur-sm">

   420→            <div className="flex items-center gap-3">
   421→              <img

   464→      {/* Mobile Bottom Navigation — shown only on mobile */}

   486→  const { pengaturan, logoSrc } = usePengaturan();
   487→  const { user } = useAuth();
   488→  const containerVariants = {
   489→    hidden: { opacity: 0 },
   490→    visible: {
   491→      opacity: 1,
   492→      transition: { staggerChildren: 0.08, delayChildren: 0.1 },
   493→    },
   494→  };
   495→
   496→  const itemVariants = {
   497→    hidden: { opacity: 0, y: 30, scale: 0.97 },
   498→    visible: {
   499→      opacity: 1,
   500→      y: 0,
   501→      scale: 1,
   502→      transition: { duration: 0.6, ease: [0.22, 1, 0.36, 1] },
   503→    },
   504→  };
   505→
   506→  // Helper: check if a view is hidden for the current user role
   507→  const isViewHiddenForUser = (viewId: string): boolean => {
   508→    const userRole = user?.role || "public";
   509→    const hiddenItems = pengaturan.sidebarConfig?.hiddenItems;
   510→    if (!hiddenItems) return false;
   511→    const roleHidden = hiddenItems[userRole] || [];
   512→    return roleHidden.includes(viewId);
   513→  };
   514→
   515→  // Quick navigation items — filtered by sidebar visibility and copilot config
   516→  const copilotEnabled = pengaturan.copilotConfig?.enabled ?? true;
   517→  const allQuickNavItems = [
   518→    { id: "ringkasan-eksekutif" as ActiveView, label: "Ringkasan Eksekutif", icon: BarChart3, color: "from-violet-500 to-purple-600", desc: "Executive Summary" },

  1143→// ============ MODERN SPLASH LOADER WRAPPER ============
  1144→function ModernSplashLoader({ isLoading, onExitComplete }: { isLoading: boolean; onExitComplete: () => void }) {
  1145→  const { pengaturan, logoSrc } = usePengaturan();
  1146→  const MIN_LOADING_MS = pengaturan.loaderDisplayTime ?? 5000;
  1147→
  1148→  // When loading completes, give the SplashLoader time to animate its exit
  1149→  // before the parent unmounts it
  1150→  useEffect(() => {
  1151→    if (!isLoading) {
  1152→      const timer = setTimeout(() => {
  1153→        onExitComplete();
  1154→      }, 1200);
  1155→      return () => clearTimeout(timer);
  1156→    }
  1157→  }, [isLoading, onExitComplete]);
  1158→
  1159→  return (
  1160→    <SplashLoader
  1161→      logoSrc={logoSrc}
  1162→      namaPemerintah={pengaturan.namaPemerintah}
  1163→      namaInstansi={pengaturan.namaInstansi}
  1164→      warnaPrimary={pengaturan.warnaPrimary}
  1165→      warnaSecondary={pengaturan.warnaSecondary}
  1166→      warnaAccent={pengaturan.warnaAccent}
  1167→      warnaDark={pengaturan.warnaDark}
  1168→      waktuTampil={MIN_LOADING_MS}
  1169→      isLoading={isLoading}
  1170→    />
  1171→  );
  1172→}
