   145→// Extract prefixes from a list of kode akun
   146→function extractPrefixesFromKodeAkun(kodeAkunList: string[]): Array<{ prefix: string; label: string; count: number }> {
   147→  const prefixCount = new Map<string, number>();
   148→
   149→  for (const kode of kodeAkunList) {
   150→    const segments = kode.split(".");
   151→    if (segments.length < 2) continue;
   152→
   153→    // Level 1: "4", "5", "6"
   154→    const l1 = segments[0];
   155→    prefixCount.set(l1, (prefixCount.get(l1) || 0) + 1);
   156→
   157→    // Level 2: "4.1", "5.1", "5.2"
   158→    if (segments.length >= 2) {
   159→      const l2 = `${segments[0]}.${segments[1]}`;
   160→      prefixCount.set(l2, (prefixCount.get(l2) || 0) + 1);
   161→    }
   162→
   163→    // Level 3: "5.1.01", "5.2.02"
   164→    if (segments.length >= 3) {
   165→      const l3 = `${segments[0]}.${segments[1]}.${segments[2]}`;
   166→      prefixCount.set(l3, (prefixCount.get(l3) || 0) + 1);
   167→    }
   168→  }
   169→
   170→  return Array.from(prefixCount.entries())
   171→    .map(([prefix, count]) => ({
   172→      prefix,
   173→      label: KODE_AKUN_PREFIXES[prefix] || `Kode ${prefix}`,
   174→      count,
   175→    }))
   176→    .sort((a, b) => a.prefix.localeCompare(b.prefix));
   177→}
